cmake_minimum_required(VERSION 3.25)
project(nookisle VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
option(NOOKISLE_REQUIRE_ALL_FEATURES "Require all optional features (libudev, libpipewire) or fail" OFF)
set(NOOKISLE_BUILD_REVISION "" CACHE STRING "Source commit SHA for manifest stamping")
add_compile_options(-ffile-prefix-map=${CMAKE_SOURCE_DIR}=.)
include(CTest)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Concurrent DBus Network Gui)
# The artwork decoder child links only this: Core and Gui, no network or D-Bus.
add_library(island_image STATIC helper/image-decode.cpp)
target_include_directories(island_image PUBLIC helper)
target_link_libraries(island_image PUBLIC Qt6::Core Qt6::Gui)
target_compile_options(island_image PRIVATE -Wall -Wextra -Wpedantic)
# Remote artwork policy and the HTTPS fetch, run only in the fetch child.
add_library(island_fetch STATIC helper/artwork-fetch.cpp)
target_link_libraries(island_fetch PUBLIC Qt6::Core Qt6::Network)
target_include_directories(island_fetch PUBLIC helper)
target_compile_options(island_fetch PRIVATE -Wall -Wextra -Wpedantic)
add_library(island_core STATIC helper/ipc-protocol.cpp helper/mpris-registry.cpp helper/artwork-loader.cpp
    helper/thumbnail-cache.cpp helper/backlight-monitor.cpp helper/system-watch.cpp)
# Backlight change events come from udev's netlink socket in the helper; without
# libudev the brightness readout stays with Omarchy's own OSD.
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
    pkg_check_modules(UDEV QUIET IMPORTED_TARGET libudev)
endif()
if(UDEV_FOUND)
    target_link_libraries(island_core PUBLIC PkgConfig::UDEV)
    target_compile_definitions(island_core PRIVATE NOOKISLE_UDEV=1)
else()
    if(NOOKISLE_REQUIRE_ALL_FEATURES)
        message(FATAL_ERROR "libudev required by NOOKISLE_REQUIRE_ALL_FEATURES but not found")
    endif()
    message(STATUS "nookisle-helper: libudev not found; the island shows no brightness readout")
endif()
option(NOOKISLE_CALENDAR "Build calendar support" ON)
if(NOOKISLE_CALENDAR)
    target_sources(island_core PRIVATE helper/calendar-parser.cpp helper/calendar-sources.cpp helper/calendar-service.cpp)
    target_compile_definitions(island_core PUBLIC NOOKISLE_CALENDAR=1)
    # Calendar sources are read and parsed on the thread pool.
    target_link_libraries(island_core PUBLIC Qt6::Concurrent)
endif()
target_include_directories(island_core PUBLIC helper)
# No Qt Gui: every image operation runs in the decoder child.
target_link_libraries(island_core PUBLIC island_fetch Qt6::Core Qt6::DBus Qt6::Network)
target_compile_options(island_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(nookisle-helper helper/main.cpp)
add_subdirectory(bridge)
target_link_libraries(nookisle-helper PRIVATE island_core island_browser)
target_compile_definitions(nookisle-helper PRIVATE ISLAND_BUILD_VERSION="${PROJECT_VERSION}")
target_compile_options(nookisle-helper PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(nookisle-helper PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
install(TARGETS nookisle-helper RUNTIME DESTINATION libexec)
add_executable(nookisle-artwork-decoder helper/artwork-decoder.cpp)
target_link_libraries(nookisle-artwork-decoder PRIVATE island_image)
target_compile_options(nookisle-artwork-decoder PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(nookisle-artwork-decoder PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
install(TARGETS nookisle-artwork-decoder RUNTIME DESTINATION libexec)
add_executable(nookisle-artwork-fetch helper/artwork-fetch-main.cpp)
target_link_libraries(nookisle-artwork-fetch PRIVATE island_fetch)
target_compile_options(nookisle-artwork-fetch PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(nookisle-artwork-fetch PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
install(TARGETS nookisle-artwork-fetch RUNTIME DESTINATION libexec)
add_library(spectrum_core STATIC helper/spectrum-core.cpp)
target_include_directories(spectrum_core PUBLIC helper)
target_compile_options(spectrum_core PRIVATE -Wall -Wextra -Wpedantic)
option(NOOKISLE_SPECTRUM "Build the PipeWire spectrum capture binary" ON)
find_package(PkgConfig QUIET)
if(NOOKISLE_SPECTRUM AND PkgConfig_FOUND)
    pkg_check_modules(PIPEWIRE QUIET IMPORTED_TARGET libpipewire-0.3)
endif()
if(PIPEWIRE_FOUND)
    add_executable(nookisle-spectrum helper/spectrum-main.cpp)
    target_link_libraries(nookisle-spectrum PRIVATE spectrum_core PkgConfig::PIPEWIRE)
    # No -Wpedantic here: SPA's inline headers use C99 idioms that pedantic C++ flags.
    target_compile_options(nookisle-spectrum PRIVATE -Wall -Wextra)
    set_target_properties(nookisle-spectrum PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    install(TARGETS nookisle-spectrum RUNTIME DESTINATION libexec)
elseif(NOOKISLE_SPECTRUM)
    if(NOOKISLE_REQUIRE_ALL_FEATURES)
        message(FATAL_ERROR "libpipewire-0.3 required by NOOKISLE_REQUIRE_ALL_FEATURES but not found")
    endif()
    message(STATUS "nookisle-spectrum: libpipewire-0.3 not found; the island shows a static play-state glyph")
else()
    if(NOOKISLE_REQUIRE_ALL_FEATURES)
        message(FATAL_ERROR "NOOKISLE_SPECTRUM disabled with NOOKISLE_REQUIRE_ALL_FEATURES")
    endif()
    message(STATUS "nookisle-spectrum: disabled; the island shows a static play-state glyph")
endif()
# Keep the source manifest usable from a source archive, and stamp the staged
# package at install time so a commit after configuration is still identified.
find_package(Git QUIET)
configure_file(cmake/install-manifest.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/install-manifest.cmake" @ONLY)
install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/install-manifest.cmake")
install(FILES BarWidget.qml Panel.qml Service.qml README.md LICENSE DESTINATION .)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/preview.png")
    install(FILES preview.png DESTINATION .)
endif()
set(NOOKISLE_SHIPPED_DOCS
    docs/install.md
    docs/settings.md
    docs/features.md
    docs/privacy.md
    docs/support-matrix.md
    docs/interface.md
)
install(FILES ${NOOKISLE_SHIPPED_DOCS} DESTINATION docs)
install(DIRECTORY qml DESTINATION . FILES_MATCHING PATTERN "*.js")
install(DIRECTORY components DESTINATION . FILES_MATCHING PATTERN "*.qml")
install(PROGRAMS scripts/write-private-shelf.py DESTINATION scripts)
install(PROGRAMS scripts/nookisle-media-keys DESTINATION libexec)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_subdirectory(tests)
    find_program(PYTHON3_EXECUTABLE python3 REQUIRED)
    add_test(NAME install-plugin COMMAND ${PYTHON3_EXECUTABLE}
        ${PROJECT_SOURCE_DIR}/tests/test-install-plugin.py)
    set_tests_properties(install-plugin PROPERTIES TIMEOUT 30)
endif()
set(CPACK_GENERATOR "TGZ")
set(CPACK_PACKAGE_NAME "nookisle")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_FILE_NAME "nookisle-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY OFF)
include(CPack)
