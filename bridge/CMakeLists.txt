add_executable(nookisle-native-host native-host.cpp)
find_package(nlohmann_json 3.11 REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
target_link_libraries(nookisle-native-host PRIVATE nlohmann_json::nlohmann_json OpenSSL::Crypto)
target_compile_options(nookisle-native-host PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(nookisle-native-host PROPERTIES AUTOMOC OFF RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
install(TARGETS nookisle-native-host RUNTIME DESTINATION libexec)
